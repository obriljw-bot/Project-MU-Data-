Attribute VB_Name = "mdl_Archive"
Option Explicit

' ========================================
' 모듈: mdl_Archive
' 설명: 월별/분기별 백업 기능
' ========================================

' ? 월별 백업
Public Sub 월별백업()
    On Error GoTo ErrorHandler
    
    Dim wsAccum As Worksheet
    Dim wbBackup As Workbook
    Dim wsBackup As Worksheet
    Dim backupFolder As String
    Dim fileName As String
    Dim lastRow As Long
    Dim yearMonth As String
    
    Set wsAccum = ThisWorkbook.Sheets("누적데이터")
    lastRow = wsAccum.Cells(wsAccum.Rows.Count, 1).End(xlUp).Row
    
    ' 데이터 확인
    If lastRow < 2 Then
        MsgBox "? 백업할 데이터가 없습니다!", vbExclamation
        Exit Sub
    End If
    
    ' 백업 폴더 생성
    backupFolder = ThisWorkbook.Path & "\백업\"
    On Error Resume Next
    If Dir(backupFolder, vbDirectory) = "" Then
        MkDir backupFolder
    End If
    On Error GoTo ErrorHandler
    
    ' 년월 추출 (데이터 기준)
    Dim firstDate As Date
    firstDate = wsAccum.Cells(2, 1).value
    yearMonth = Format(firstDate, "yyyy년_mm월")
    
    fileName = backupFolder & yearMonth & ".xlsx"
    
    ' 파일 존재 확인
    If Dir(fileName) <> "" Then
        Dim response As VbMsgBoxResult
        response = MsgBox("백업 파일이 이미 존재합니다:" & vbCrLf & vbCrLf & _
                          fileName & vbCrLf & vbCrLf & _
                          "덮어쓰시겠습니까?", vbYesNo + vbQuestion, "파일 존재")
        
        If response = vbNo Then Exit Sub
        
        Kill fileName
    End If
    
    OptimizePerformance True
    
    ' 새 백업 파일 생성
    Set wbBackup = Workbooks.Add
    Set wsBackup = wbBackup.Sheets(1)
    wsBackup.Name = "누적데이터"
    
    ' 데이터 복사
    wsAccum.Range("A1:L" & lastRow).Copy wsBackup.Range("A1")
    Application.CutCopyMode = False
    
    ' 서식 복사
    With wsBackup
        .Range("A1:L1").HorizontalAlignment = xlCenter
        .Range("A1:L1").Font.Bold = True
        .Range("A1:L1").Interior.color = RGB(68, 114, 196)
        .Range("A1:L1").Font.color = RGB(255, 255, 255)
        
        .Range("A:A").NumberFormat = "yyyy-mm-dd"
        .Range("F:F").NumberFormat = "0"
        .Range("G:G").NumberFormat = "#,##0"
        .Range("H:I").NumberFormat = "0"
        .Range("J:K").NumberFormat = "#,##0"
        
        .Cells.EntireColumn.AutoFit
    End With
    
    ' 파일 저장
    Application.DisplayAlerts = False
    wbBackup.SaveAs fileName, FileFormat:=xlOpenXMLWorkbook
    wbBackup.Close False
    Application.DisplayAlerts = True
    
    ' 누적데이터 초기화 여부 확인
    Dim clearResponse As VbMsgBoxResult
    clearResponse = MsgBox("? 백업 완료!" & vbCrLf & vbCrLf & _
                           "?? " & fileName & vbCrLf & _
                           "?? " & (lastRow - 1) & " 행 백업됨" & vbCrLf & vbCrLf & _
                           "누적데이터 시트를 초기화하시겠습니까?", _
                           vbYesNo + vbQuestion, "백업 완료")
    
    If clearResponse = vbYes Then
        ' 헤더만 남기고 데이터 삭제
        If lastRow > 1 Then
            wsAccum.Range("A2:L" & lastRow).ClearContents
        End If
        
        MsgBox "? 누적데이터가 초기화되었습니다!" & vbCrLf & vbCrLf & _
               "새로운 데이터를 취합할 수 있습니다.", vbInformation, "초기화 완료"
    End If
    
    OptimizePerformance False
    
    Exit Sub
    
ErrorHandler:
    OptimizePerformance False
    MsgBox "백업 오류: " & Err.Description, vbCritical
End Sub

' ? 백업 파일 목록 조회
Public Function GetBackupFiles() As Collection
    Dim backupFolder As String
    Dim fileName As String
    Dim fileList As Collection
    
    Set fileList = New Collection
    
    backupFolder = ThisWorkbook.Path & "\백업\"
    
    If Dir(backupFolder, vbDirectory) = "" Then
        Set GetBackupFiles = fileList
        Exit Function
    End If
    
    fileName = Dir(backupFolder & "*.xlsx")
    
    Do While fileName <> ""
        fileList.Add fileName
        fileName = Dir()
    Loop
    
    Set GetBackupFiles = fileList
End Function

' 성능 최적화
Private Sub OptimizePerformance(enable As Boolean)
    With Application
        .ScreenUpdating = Not enable
        .EnableEvents = Not enable
        .Calculation = IIf(enable, xlCalculationManual, xlCalculationAutomatic)
        .DisplayAlerts = Not enable
    End With
End Sub
