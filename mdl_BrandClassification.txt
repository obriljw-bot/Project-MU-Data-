' ??? 메인 함수: 누적데이터에서 브랜드 파일 생성
Public Sub 브랜드파일생성_누적데이터(selectedBrands As Collection)
    On Error GoTo ErrorHandler
    
    Dim wsSource As Worksheet
    Dim folderPath As String
    Dim todayDate As String
    Dim processedBrands As Object
    Dim integratedBrands As Collection
    Dim brandItem As Object
    Dim remainingBrands As Object
    Dim brandName As Variant
    Dim i As Long
    Dim isAllBrands As Boolean
    Dim processedCount As Long
    
    Debug.Print vbCrLf & "=== 브랜드파일생성_누적데이터 시작 ==="
    
    isAllBrands = (selectedBrands.Count = 1 And selectedBrands(1) = "*ALL*")
    
    OptimizePerformance True
    
    ' ? 누적데이터 시트 사용
    On Error Resume Next
    Set wsSource = ThisWorkbook.Sheets("누적데이터")
    On Error GoTo ErrorHandler
    
    If wsSource Is Nothing Then
        MsgBox "? 누적데이터 시트가 없습니다!", vbCritical
        GoTo CleanExit
    End If
    
    Dim lastRow As Long
    lastRow = wsSource.Cells(wsSource.Rows.Count, 3).End(xlUp).Row
    
    Debug.Print "누적데이터 행 수: " & (lastRow - 1)
    
    If lastRow < 2 Then
        MsgBox "? 누적데이터가 비어있습니다!", vbCritical
        GoTo CleanExit
    End If
    
    ' 경로 설정
    todayDate = Format(Date, "yymmdd")
    folderPath = GetSetting(SET_기본경로)
    
    Debug.Print "기본 경로: " & folderPath
    
    If folderPath = "" Then
        MsgBox "? 저장 경로가 설정되지 않았습니다!" & vbCrLf & vbCrLf & _
               "시스템설정 시트에서 '기본저장경로'를 설정하세요.", vbCritical
        GoTo CleanExit
    End If
    
    If Right(folderPath, 1) <> "\" Then folderPath = folderPath & "\"
    folderPath = folderPath & todayDate
    
    Debug.Print "저장 폴더: " & folderPath
    
    ' 폴더 생성
    On Error Resume Next
    If Dir(folderPath, vbDirectory) = "" Then
        Dim basePath As String
        basePath = GetSetting(SET_기본경로)
        If Right(basePath, 1) <> "\" Then basePath = basePath & "\"
        If Dir(basePath, vbDirectory) = "" Then MkDir basePath
        MkDir folderPath
        Debug.Print "폴더 생성됨: " & folderPath
    Else
        Debug.Print "폴더 이미 존재: " & folderPath
    End If
    On Error GoTo ErrorHandler
    
    Set processedBrands = CreateObject("Scripting.Dictionary")
    processedCount = 0
    
    ' 통합 브랜드 처리
    Set integratedBrands = LoadIntegratedBrands()
    Debug.Print "통합 브랜드 개수: " & integratedBrands.Count
    
    If integratedBrands.Count > 0 Then
        For i = 1 To integratedBrands.Count
            Set brandItem = integratedBrands(i)
            
            If isAllBrands Or IsBrandSelected(brandItem("Name"), selectedBrands) Then
                Debug.Print "처리 중: " & brandItem("Name")
                
                If GetSetting(SET_진행표시) = "O" Then
                    UpdateProgress "브랜드 파일 생성", brandItem("Name"), Int(processedCount / (integratedBrands.Count + 10) * 100)
                End If
                
                ProcessIntegratedBrand_FromAccumulated wsSource, brandItem, folderPath, todayDate, processedBrands
                processedCount = processedCount + 1
            End If
        Next i
    End If
    
    ' 개별 브랜드 처리
    Set remainingBrands = GetRemainingBrands(wsSource, processedBrands)
    Debug.Print "개별 브랜드 개수: " & remainingBrands.Count
    
    For Each brandName In remainingBrands.Keys
        If isAllBrands Or IsBrandSelected(CStr(brandName), selectedBrands) Then
            Debug.Print "처리 중: " & brandName
            
            If GetSetting(SET_진행표시) = "O" Then
                UpdateProgress "브랜드 파일 생성", CStr(brandName), Int(processedCount / (integratedBrands.Count + remainingBrands.Count) * 100)
            End If
            
            ProcessBrandData_FromAccumulated wsSource, CStr(brandName), folderPath, todayDate
            processedCount = processedCount + 1
        End If
    Next brandName
    
CleanExit:
    OptimizePerformance False
    Application.CutCopyMode = False
    
    Debug.Print "처리 완료 개수: " & processedCount
    Debug.Print "=== 브랜드파일생성_누적데이터 종료 ===" & vbCrLf
    
    If processedCount > 0 Then
        MsgBox "? 브랜드 파일 생성 완료!" & vbCrLf & vbCrLf & _
               "생성 브랜드: " & processedCount & "개" & vbCrLf & _
               "?? 저장 위치: " & folderPath & "\", vbInformation, "완료"
    Else
        MsgBox "?? 생성된 파일이 없습니다!" & vbCrLf & vbCrLf & _
               "선택한 브랜드에 해당하는 데이터가 없거나," & vbCrLf & _
               "브랜드 설정을 확인하세요.", vbExclamation, "알림"
    End If
    
    Exit Sub
    
ErrorHandler:
    Debug.Print "? 오류: " & Err.Description
    MsgBox "오류: " & Err.Description, vbCritical
    Resume CleanExit
End Sub

' ? 헬퍼: 브랜드 선택 확인
Private Function IsBrandSelected(brandName As String, selectedBrands As Collection) As Boolean
    Dim item As Variant
    For Each item In selectedBrands
        If CStr(item) = brandName Then
            IsBrandSelected = True
            Exit Function
        End If
    Next item
    IsBrandSelected = False
End Function

' ? 통합 브랜드 처리 (누적데이터 기반)
Private Sub ProcessIntegratedBrand_FromAccumulated(wsSource As Worksheet, brandItem As Object, _
                                                   folderPath As String, todayDate As String, _
                                                   processedBrands As Object)
    On Error GoTo ErrorHandler
    
    Dim wbNew As Workbook
    Dim wsNew As Worksheet
    Dim fileName As String
    Dim brandName As String
    Dim brands As Collection
    Dim i As Long, Row As Long
    Dim lastRow As Long
    Dim pasteRow As Long
    Dim cellBrand As String
    Dim copyCount As Long
    
    brandName = brandItem("Name")
    Set brands = brandItem("Brands")
    
    Debug.Print "  통합브랜드 처리: " & brandName & " (하위: " & brands.Count & "개)"
    
    For i = 1 To brands.Count
        processedBrands(brands(i)) = True
    Next i
    
    Set wbNew = Workbooks.Add
    Set wsNew = wbNew.Sheets(1)
    wsNew.Name = "판매데이터"
    
    ' 서식 설정
    wsNew.Range("D:E").NumberFormat = "0"
    
    ' 헤더
    wsNew.Cells(1, 1).value = "판매일"
    wsNew.Cells(1, 2).value = "판매매장"
    wsNew.Cells(1, 3).value = "브랜드"
    wsNew.Cells(1, 4).value = "코드"
    wsNew.Cells(1, 5).value = "이름"
    wsNew.Cells(1, 6).value = "판매수량"
    wsNew.Cells(1, 7).value = "판매금액"
    
    With wsNew.Range("A1:G1")
        .HorizontalAlignment = xlCenter
        .Font.Bold = True
        .Interior.color = RGB(68, 114, 196)
        .Font.color = RGB(255, 255, 255)
    End With
    
    ' 데이터 복사
    lastRow = wsSource.Cells(wsSource.Rows.Count, 3).End(xlUp).Row
    pasteRow = 2
    copyCount = 0
    
    For Row = 2 To lastRow
        cellBrand = Trim(wsSource.Cells(Row, 3).value)
        
        For i = 1 To brands.Count
            If cellBrand = brands(i) Then
                wsNew.Cells(pasteRow, 1).value = wsSource.Cells(Row, 1).value  ' 판매일
                wsNew.Cells(pasteRow, 2).value = wsSource.Cells(Row, 2).value  ' 판매매장
                wsNew.Cells(pasteRow, 3).value = wsSource.Cells(Row, 3).value  ' 브랜드
                wsNew.Cells(pasteRow, 4).value = wsSource.Cells(Row, 4).value  ' 코드
                wsNew.Cells(pasteRow, 5).value = wsSource.Cells(Row, 5).value  ' 이름
                wsNew.Cells(pasteRow, 6).value = wsSource.Cells(Row, 8).value  ' 판매수량
                wsNew.Cells(pasteRow, 7).value = wsSource.Cells(Row, 9).value  ' 판매금액
                
                pasteRow = pasteRow + 1
                copyCount = copyCount + 1
                Exit For
            End If
        Next i
    Next Row
    
    Debug.Print "    복사된 행: " & copyCount
    
    If copyCount = 0 Then
        wbNew.Close False
        Debug.Print "    데이터 없음 - 파일 생성 안함"
        Exit Sub
    End If
    
    ' 서식
    With wsNew
        .Range("A:G").HorizontalAlignment = xlCenter
        .Range("D:E").HorizontalAlignment = xlLeft
        .Range("A2:A" & pasteRow - 1).NumberFormat = "yyyy-mm-dd"
        .Range("F2:F" & pasteRow - 1).NumberFormat = "0"
        .Range("G2:G" & pasteRow - 1).NumberFormat = "#,##0"
        .Cells.EntireColumn.AutoFit
    End With
    
    ' 요약 시트
    If GetSetting(SET_요약시트) = "O" Then
        On Error Resume Next
        Call CreateBrandSummary(wbNew, brandName)
        On Error GoTo ErrorHandler
    End If
    
    ' 파일 저장
    fileName = folderPath & "\" & GetSetting(SET_파일접두어) & brandName & _
               GetSetting(SET_파일접미어) & todayDate & ".xlsx"
    
    Debug.Print "    저장: " & fileName
    
    wbNew.SaveAs fileName, FileFormat:=xlOpenXMLWorkbook
    wbNew.Close False
    
    Exit Sub
    
ErrorHandler:
    Debug.Print "    ? 오류: " & Err.Description
    If Not wbNew Is Nothing Then wbNew.Close False
End Sub

' ? 일자 필터 버전: 개별 브랜드
Private Sub ProcessBrandData_Filtered(wsSource As Worksheet, brandName As String, _
                                      folderPath As String, todayDate As String, _
                                      startDate As Date, endDate As Date)
    On Error GoTo ErrorHandler
    
    Dim wbNew As Workbook
    Dim wsNew As Worksheet
    Dim fileName As String
    Dim lastRow As Long
    Dim i As Long
    Dim pasteRow As Long
    Dim copyCount As Long
    Dim saleDate As Date
    
    Set wbNew = Workbooks.Add
    Set wsNew = wbNew.Sheets(1)
    wsNew.Name = "판매데이터"
    
    wsNew.Range("D:E").NumberFormat = "0"
    
    ' 헤더
    wsNew.Cells(1, 1).value = "판매일"
    wsNew.Cells(1, 2).value = "판매매장"
    wsNew.Cells(1, 3).value = "브랜드"
    wsNew.Cells(1, 4).value = "코드"
    wsNew.Cells(1, 5).value = "이름"
    wsNew.Cells(1, 6).value = "판매수량"
    wsNew.Cells(1, 7).value = "판매금액"
    
    With wsNew.Range("A1:G1")
        .HorizontalAlignment = xlCenter
        .Font.Bold = True
        .Interior.color = RGB(68, 114, 196)
        .Font.color = RGB(255, 255, 255)
    End With
    
    ' 데이터 복사 (일자 필터)
    lastRow = wsSource.Cells(wsSource.Rows.Count, 3).End(xlUp).Row
    pasteRow = 2
    copyCount = 0
    
    For i = 2 To lastRow
        On Error Resume Next
        saleDate = wsSource.Cells(i, 1).value
        On Error GoTo ErrorHandler
        
        If saleDate >= startDate And saleDate <= endDate Then
            If Trim(wsSource.Cells(i, 3).value) = brandName Then
                wsNew.Cells(pasteRow, 1).value = wsSource.Cells(i, 1).value
                wsNew.Cells(pasteRow, 2).value = wsSource.Cells(i, 2).value
                wsNew.Cells(pasteRow, 3).value = wsSource.Cells(i, 3).value
                wsNew.Cells(pasteRow, 4).value = wsSource.Cells(i, 4).value
                wsNew.Cells(pasteRow, 5).value = wsSource.Cells(i, 5).value
                wsNew.Cells(pasteRow, 6).value = wsSource.Cells(i, 8).value
                wsNew.Cells(pasteRow, 7).value = wsSource.Cells(i, 9).value
                
                pasteRow = pasteRow + 1
                copyCount = copyCount + 1
            End If
        End If
    Next i
    
    If copyCount = 0 Then
        wbNew.Close False
        Exit Sub
    End If
    
    ' 서식
    With wsNew
        .Range("A:G").HorizontalAlignment = xlCenter
        .Range("D:E").HorizontalAlignment = xlLeft
        .Range("A2:A" & pasteRow - 1).NumberFormat = "yyyy-mm-dd"
        .Range("F2:F" & pasteRow - 1).NumberFormat = "0"
        .Range("G2:G" & pasteRow - 1).NumberFormat = "#,##0"
        .Cells.EntireColumn.AutoFit
    End With
    
    If GetSetting(SET_요약시트) = "O" Then
        On Error Resume Next
        Call CreateBrandSummary(wbNew, brandName)
        On Error GoTo ErrorHandler
    End If
    
    fileName = folderPath & "\" & GetSetting(SET_파일접두어) & brandName & _
               GetSetting(SET_파일접미어) & todayDate & ".xlsx"
    
    wbNew.SaveAs fileName, FileFormat:=xlOpenXMLWorkbook
    wbNew.Close False
    
    Exit Sub
    
ErrorHandler:
    If Not wbNew Is Nothing Then wbNew.Close False
End Sub

' ? 남은 브랜드 목록
Private Function GetRemainingBrands(ws As Worksheet, processedBrands As Object) As Object
    Dim result As Object
    Dim lastRow As Long
    Dim i As Long
    Dim brandName As String
    
    Set result = CreateObject("Scripting.Dictionary")
    lastRow = ws.Cells(ws.Rows.Count, 3).End(xlUp).Row
    
    For i = 2 To lastRow
        brandName = Trim(ws.Cells(i, 3).value)
        
        If brandName <> "" Then
            If Not processedBrands.exists(brandName) And Not result.exists(brandName) Then
                result.Add brandName, Nothing
            End If
        End If
    Next i
    
    Set GetRemainingBrands = result
End Function

' 성능 최적화
Private Sub OptimizePerformance(enable As Boolean)
    With Application
        .ScreenUpdating = Not enable
        .EnableEvents = Not enable
        .Calculation = IIf(enable, xlCalculationManual, xlCalculationAutomatic)
        .DisplayAlerts = Not enable
    End With
End Sub

' 진행 상황 업데이트
Private Sub UpdateProgress(stepName As String, itemName As String, percent As Long)
    On Error Resume Next
    If Not frmProgress Is Nothing Then
        frmProgress.UpdateProgress stepName, itemName, percent
    Else
        Application.StatusBar = stepName & ": " & percent & "% (" & itemName & ")"
    End If
End Sub

' 요약 시트 생성 (기존 버전 복원)
Private Sub CreateBrandSummary(wb As Workbook, brandName As String)
    On Error Resume Next
    
    Dim wsSummary As Worksheet
    Dim wsData As Worksheet
    Dim lastRow As Long
    Dim i As Long
    Dim currentRow As Long
    
    Set wsData = wb.Sheets(1)
    Set wsSummary = wb.Sheets.Add(Before:=wsData)
    wsSummary.Name = "요약"
    
    lastRow = wsData.Cells(wsData.Rows.Count, 1).End(xlUp).Row
    
    With wsSummary
        ' 헤더
        .Range("A1:H1").Merge
        .Range("A1").value = brandName & " 판매 요약"
        .Range("A1").Font.Size = 16
        .Range("A1").Font.Bold = True
        .Range("A1").Font.color = RGB(255, 255, 255)
        .Range("A1").Interior.color = RGB(68, 114, 196)
        .Range("A1").HorizontalAlignment = xlCenter
        .Range("A1").VerticalAlignment = xlCenter
        .Rows(1).RowHeight = 35
        
        ' === 브랜드별 판매 데이터 (통합 브랜드인 경우 가로형) ===
        Dim brandDict As Object
        Set brandDict = CreateObject("Scripting.Dictionary")
        Dim brandCell As Range
        
        For Each brandCell In wsData.Range("C2:C" & lastRow)
            If Not brandDict.exists(brandCell.value) And brandCell.value <> "" Then
                brandDict.Add brandCell.value, Nothing
            End If
        Next brandCell
        
        ' 브랜드가 2개 이상이면 통합 브랜드
        If brandDict.Count > 1 Then
            Dim brandNames As Variant
            brandNames = brandDict.Keys
            Dim totalCol As Long
            totalCol = brandDict.Count + 2  ' 합계 컬럼 위치
            
            ' 요약 박스 병합
            .Range("A3:" & Chr(65 + totalCol - 1) & "3").Merge
            .Range("A3").value = "판매 실적 요약"
            .Range("A3").Font.Bold = True
            .Range("A3").Font.Size = 11
            .Range("A3").Interior.color = RGB(217, 225, 242)
            .Range("A3").HorizontalAlignment = xlLeft
            
            ' 가로형 레이아웃: A4, B4, C4... 에 브랜드명
            .Cells(4, 1).value = "브랜드"
            .Cells(4, 1).Font.Bold = True
            
            For i = 0 To brandDict.Count - 1
                If i + 2 <= 20 Then  ' 최대 컬럼 제한
                    .Cells(4, i + 2).value = brandNames(i)
                    .Cells(4, i + 2).Font.Bold = True
                End If
            Next i
            
            ' 합계 컬럼 추가
            .Cells(4, totalCol).value = "합계"
            .Cells(4, totalCol).Font.Bold = True
            .Cells(4, totalCol).Font.color = RGB(192, 0, 0)
            
            ' 총 판매수량
            .Cells(5, 1).value = "총 판매수량"
            For i = 0 To brandDict.Count - 1
                If i + 2 <= 20 Then
                    .Cells(5, i + 2).Formula = "=SUMIF(판매데이터!$C$2:$C$" & lastRow & ",""" & brandNames(i) & """,판매데이터!$F$2:$F$" & lastRow & ")"
                    .Cells(5, i + 2).NumberFormat = "#,##0 개"
                End If
            Next i
            
            ' 총 판매수량 합계
            .Cells(5, totalCol).Formula = "=SUM(B5:" & Chr(65 + brandDict.Count) & "5)"
            .Cells(5, totalCol).NumberFormat = "#,##0 개"
            .Cells(5, totalCol).Font.Bold = True
            .Cells(5, totalCol).Font.color = RGB(0, 112, 192)
            
            ' 총 판매금액
            .Cells(6, 1).value = "총 판매금액"
            For i = 0 To brandDict.Count - 1
                If i + 2 <= 20 Then
                    .Cells(6, i + 2).Formula = "=SUMIF(판매데이터!$C$2:$C$" & lastRow & ",""" & brandNames(i) & """,판매데이터!$G$2:$G$" & lastRow & ")"
                    .Cells(6, i + 2).NumberFormat = "#,##0 원"
                    .Cells(6, i + 2).Font.color = RGB(192, 0, 0)
                End If
            Next i
            
            ' 총 판매금액 합계
            .Cells(6, totalCol).Formula = "=SUM(B6:" & Chr(65 + brandDict.Count) & "6)"
            .Cells(6, totalCol).NumberFormat = "#,##0 원"
            .Cells(6, totalCol).Font.Bold = True
            .Cells(6, totalCol).Font.color = RGB(192, 0, 0)
            
            ' 테두리 (합계 컬럼 포함)
            .Range("A4:" & Chr(65 + totalCol - 1) & "6").Borders.LineStyle = xlContinuous
            .Range("A4:" & Chr(65 + totalCol - 1) & "6").Borders.Weight = xlThin
            .Range("A4:" & Chr(65 + totalCol - 1) & "6").Borders.color = RGB(68, 114, 196)
            
            ' 다음 섹션 시작 위치 조정
            currentRow = 8
        Else
            ' 개별 브랜드: 기존 방식
            .Range("A3:D3").Merge
            .Range("A3").value = "판매 실적 요약"
            .Range("A3").Font.Bold = True
            .Range("A3").Font.Size = 11
            .Range("A3").Interior.color = RGB(217, 225, 242)
            .Range("A3").HorizontalAlignment = xlLeft
            
            .Cells(4, 1).value = "총 판매수량"
            .Cells(4, 2).Formula = "=SUM(판매데이터!F2:F" & lastRow & ")"
            .Cells(4, 2).NumberFormat = "#,##0 개"
            .Cells(4, 2).Font.Bold = True
            .Cells(4, 2).Font.Size = 12
            .Cells(4, 2).Font.color = RGB(0, 112, 192)
            
            .Cells(5, 1).value = "총 판매금액"
            .Cells(5, 2).Formula = "=SUM(판매데이터!G2:G" & lastRow & ")"
            .Cells(5, 2).NumberFormat = "#,##0 원"
            .Cells(5, 2).Font.Bold = True
            .Cells(5, 2).Font.Size = 12
            .Cells(5, 2).Font.color = RGB(192, 0, 0)
            
            .Range("A3:D5").Borders.LineStyle = xlContinuous
            .Range("A3:D5").Borders.Weight = xlThin
            .Range("A3:D5").Borders.color = RGB(68, 114, 196)
            
            currentRow = 7
        End If
        
        ' === 일자별 판매 현황 ===
        .Cells(currentRow, 1).value = "일자별 판매 현황"
        .Cells(currentRow, 1).Font.Bold = True
        .Cells(currentRow, 1).Font.Size = 11
        .Range(.Cells(currentRow, 1), .Cells(currentRow, 3)).Interior.color = RGB(217, 225, 242)
        currentRow = currentRow + 1
        
        .Cells(currentRow, 1).value = "판매일"
        .Cells(currentRow, 2).value = "판매수량"
        .Cells(currentRow, 3).value = "판매금액"
        .Range(.Cells(currentRow, 1), .Cells(currentRow, 3)).Font.Bold = True
        .Range(.Cells(currentRow, 1), .Cells(currentRow, 3)).Interior.color = RGB(68, 114, 196)
        .Range(.Cells(currentRow, 1), .Cells(currentRow, 3)).Font.color = RGB(255, 255, 255)
        .Range(.Cells(currentRow, 1), .Cells(currentRow, 3)).HorizontalAlignment = xlCenter
        currentRow = currentRow + 1
        
        Dim dateStartRow As Long
        dateStartRow = currentRow
        
        Dim dateDict As Object
        Dim cell As Range
        Set dateDict = CreateObject("Scripting.Dictionary")
        
        For Each cell In wsData.Range("A2:A" & lastRow)
            If Not dateDict.exists(cell.value) And cell.value <> "" Then
                dateDict.Add cell.value, Nothing
            End If
        Next cell
        
        Dim dates As Variant
        dates = dateDict.Keys
        
        For i = 0 To dateDict.Count - 1
            .Cells(currentRow + i, 1).value = dates(i)
            .Cells(currentRow + i, 1).NumberFormat = "yyyy-mm-dd"
            .Cells(currentRow + i, 2).Formula = "=SUMIF(판매데이터!$A$2:$A$" & lastRow & "," & _
                .Cells(currentRow + i, 1).Address & ",판매데이터!$F$2:$F$" & lastRow & ")"
            .Cells(currentRow + i, 2).NumberFormat = "#,##0"
            .Cells(currentRow + i, 3).Formula = "=SUMIF(판매데이터!$A$2:$A$" & lastRow & "," & _
                .Cells(currentRow + i, 1).Address & ",판매데이터!$G$2:$G$" & lastRow & ")"
            .Cells(currentRow + i, 3).NumberFormat = "#,##0"
            
            If i Mod 2 = 0 Then
                .Range(.Cells(currentRow + i, 1), .Cells(currentRow + i, 3)).Interior.color = RGB(242, 242, 242)
            End If
        Next i
        
        .Range(.Cells(dateStartRow - 1, 1), .Cells(currentRow + dateDict.Count - 1, 3)).Borders.LineStyle = xlContinuous
        
        ' === 매장별 판매 현황 ===
        currentRow = IIf(brandDict.Count > 1, 8, 7)
        
        .Cells(currentRow, 5).value = "매장별 판매 현황"
        .Cells(currentRow, 5).Font.Bold = True
        .Cells(currentRow, 5).Font.Size = 11
        .Range(.Cells(currentRow, 5), .Cells(currentRow, 7)).Interior.color = RGB(217, 225, 242)
        currentRow = currentRow + 1
        
        .Cells(currentRow, 5).value = "판매매장"
        .Cells(currentRow, 6).value = "판매수량"
        .Cells(currentRow, 7).value = "판매금액"
        .Range(.Cells(currentRow, 5), .Cells(currentRow, 7)).Font.Bold = True
        .Range(.Cells(currentRow, 5), .Cells(currentRow, 7)).Interior.color = RGB(68, 114, 196)
        .Range(.Cells(currentRow, 5), .Cells(currentRow, 7)).Font.color = RGB(255, 255, 255)
        .Range(.Cells(currentRow, 5), .Cells(currentRow, 7)).HorizontalAlignment = xlCenter
        currentRow = currentRow + 1
        
        Dim storeStartRow As Long
        storeStartRow = currentRow
        
        Dim storeDict As Object
        Set storeDict = CreateObject("Scripting.Dictionary")
        
        For Each cell In wsData.Range("B2:B" & lastRow)
            If Not storeDict.exists(cell.value) And cell.value <> "" Then
                storeDict.Add cell.value, Nothing
            End If
        Next cell
        
        Dim stores As Variant
        stores = storeDict.Keys
        
        For i = 0 To storeDict.Count - 1
            .Cells(currentRow + i, 5).value = stores(i)
            .Cells(currentRow + i, 6).Formula = "=SUMIF(판매데이터!$B$2:$B$" & lastRow & ",""" & stores(i) & """,판매데이터!$F$2:$F$" & lastRow & ")"
            .Cells(currentRow + i, 6).NumberFormat = "#,##0"
            .Cells(currentRow + i, 7).Formula = "=SUMIF(판매데이터!$B$2:$B$" & lastRow & ",""" & stores(i) & """,판매데이터!$G$2:$G$" & lastRow & ")"
            .Cells(currentRow + i, 7).NumberFormat = "#,##0"
            
            If i Mod 2 = 0 Then
                .Range(.Cells(currentRow + i, 5), .Cells(currentRow + i, 7)).Interior.color = RGB(242, 242, 242)
            End If
        Next i
        
        .Range(.Cells(storeStartRow - 1, 5), .Cells(currentRow + storeDict.Count - 1, 7)).Borders.LineStyle = xlContinuous
        
        ' 스타일
        .Range(.Cells(dateStartRow, 1), .Cells(currentRow + dateDict.Count - 1, 3)).HorizontalAlignment = xlCenter
        .Range(.Cells(storeStartRow, 5), .Cells(currentRow + storeDict.Count - 1, 7)).HorizontalAlignment = xlCenter
        
        .Columns("A:H").AutoFit
        
        For i = 1 To 8
            If .Columns(i).ColumnWidth < 12 Then
                .Columns(i).ColumnWidth = 12
            End If
        Next i
        
        .Rows("3:1000").RowHeight = 20
        ActiveWindow.DisplayGridlines = False
    End With
    
    On Error GoTo 0
End Sub
