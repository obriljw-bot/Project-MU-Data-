' 시스템 설정값 가져오기
Public Function GetSetting(settingName As String) As String
    On Error GoTo ErrorHandler
    
    Dim ws As Worksheet
    Dim lastRow As Long
    Dim i As Long
    Dim sheetFound As Boolean
    Dim foundValue As String
    
    ' 시트 찾기
    sheetFound = False
    
    On Error Resume Next
    Set ws = ThisWorkbook.Sheets("시스템설정")
    If Not ws Is Nothing Then sheetFound = True
    On Error GoTo ErrorHandler
    
    If Not sheetFound Then
        On Error Resume Next
        Set ws = ThisWorkbook.Sheets("시스템 설정")
        If Not ws Is Nothing Then sheetFound = True
        On Error GoTo ErrorHandler
    End If
    
    If Not sheetFound Then
        On Error Resume Next
        Set ws = ThisWorkbook.Sheets(SHEET_시스템설정)
        If Not ws Is Nothing Then sheetFound = True
        On Error GoTo ErrorHandler
    End If
    
    If Not sheetFound Or ws Is Nothing Then
        GoTo DefaultValue
    End If
    
    On Error GoTo DefaultValue
    
    lastRow = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row
    
    ' 시트에서 설정값 찾기
    For i = 2 To lastRow
        If Trim(ws.Cells(i, 1).value) = settingName Then
            foundValue = ws.Cells(i, 2).value
            GetSetting = foundValue
            Exit Function
        End If
    Next i
    
DefaultValue:
    ' 설정이 없으면 기본값
    Select Case settingName
        Case SET_기본경로, "기본저장경로"
            GetSetting = ""
        Case SET_파일접두어
            GetSetting = FILE_PREFIX
        Case SET_파일접미어
            GetSetting = FILE_SUFFIX
        Case SET_요약시트, SET_중복검증, SET_진행표시
            GetSetting = "O"
        Case Else
            GetSetting = ""
    End Select
    Exit Function
    
ErrorHandler:
    GoTo DefaultValue
End Function

' 매장 목록 가져오기
Public Function GetStoreList() As Collection
    On Error GoTo ErrorHandler
    
    Dim ws As Worksheet
    Dim lastRow As Long
    Dim i As Long
    Dim result As New Collection
    Dim storeValue As Variant
    Dim useFlag As Variant
    Dim sheetFound As Boolean
    
    ' 시트 찾기
    sheetFound = False
    
    On Error Resume Next
    Set ws = ThisWorkbook.Sheets("매장설정")
    If Not ws Is Nothing Then sheetFound = True
    On Error GoTo ErrorHandler
    
    If Not sheetFound Then
        On Error Resume Next
        Set ws = ThisWorkbook.Sheets("매장 설정")
        If Not ws Is Nothing Then sheetFound = True
        On Error GoTo ErrorHandler
    End If
    
    If Not sheetFound Then
        On Error Resume Next
        Set ws = ThisWorkbook.Sheets(SHEET_매장설정)
        If Not ws Is Nothing Then sheetFound = True
        On Error GoTo ErrorHandler
    End If
    
    If Not sheetFound Or ws Is Nothing Then
        Set GetStoreList = result
        Exit Function
    End If
    
    lastRow = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row
    
    ' 데이터 읽기
    For i = 2 To lastRow
        storeValue = ws.Cells(i, 2).value ' B열: 매장명
        useFlag = ws.Cells(i, 6).value    ' F열: 사용여부
        
        ' 사용여부 체크 (공백 제거)
        If Trim(UCase(CStr(useFlag))) = "O" Then
            If storeValue <> "" Then
                result.Add CStr(storeValue)
            End If
        End If
    Next i
    
    Set GetStoreList = result
    Exit Function
    
ErrorHandler:
    Set GetStoreList = New Collection
End Function

' 통합 브랜드 로드
Public Function LoadIntegratedBrands() As Collection
    On Error GoTo ErrorHandler
    
    Dim ws As Worksheet
    Dim lastRow As Long
    Dim i As Long, j As Long
    Dim result As New Collection
    Dim brandItem As Object
    Dim subBrands As Collection
    Dim brandName As String
    Dim sheetFound As Boolean
    Dim useFlag As String
    
    ' 디버그 모드
    Dim debugMode As Boolean
    debugMode = True
    
    If debugMode Then
        Debug.Print vbCrLf & "=== LoadIntegratedBrands 시작 ==="
    End If
    
    ' 시트 찾기
    sheetFound = False
    
    On Error Resume Next
    Set ws = ThisWorkbook.Sheets("브랜드설정")
    If Not ws Is Nothing Then
        sheetFound = True
        If debugMode Then Debug.Print "? 시트 발견: 브랜드설정"
    End If
    On Error GoTo ErrorHandler
    
    If Not sheetFound Then
        On Error Resume Next
        Set ws = ThisWorkbook.Sheets("브랜드 설정")
        If Not ws Is Nothing Then
            sheetFound = True
            If debugMode Then Debug.Print "? 시트 발견: 브랜드 설정"
        End If
        On Error GoTo ErrorHandler
    End If
    
    If Not sheetFound Then
        On Error Resume Next
        Set ws = ThisWorkbook.Sheets(SHEET_브랜드설정)
        If Not ws Is Nothing Then
            sheetFound = True
            If debugMode Then Debug.Print "? 시트 발견: " & SHEET_브랜드설정
        End If
        On Error GoTo ErrorHandler
    End If
    
    If Not sheetFound Or ws Is Nothing Then
        If debugMode Then Debug.Print "? 시트를 찾을 수 없음!"
        Set LoadIntegratedBrands = result
        Exit Function
    End If
    
    lastRow = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row
    If debugMode Then Debug.Print "마지막 행: " & lastRow
    
    ' 통합 브랜드 읽기
    For i = 2 To lastRow
        brandName = Trim(CStr(ws.Cells(i, 1).value))
        useFlag = Trim(UCase(CStr(ws.Cells(i, 7).value)))
        
        If debugMode Then
            Debug.Print ""
            Debug.Print i & "행 처리 중..."
            Debug.Print "  A열(통합브랜드명): [" & brandName & "]"
            Debug.Print "  G열(사용여부): [" & useFlag & "]"
        End If
        
        ' G열(7): 사용여부가 "O"인 경우만
        If useFlag = "O" Then
            If debugMode Then Debug.Print "  → 사용여부 확인: ?"
            
            If brandName <> "" Then
                If debugMode Then Debug.Print "  → 브랜드명 확인: ?"
                
                ' ?? 핵심 수정: Dictionary 대신 새로운 객체 생성
                Set brandItem = CreateObject("Scripting.Dictionary")
                Set subBrands = New Collection
                
                ' B~F열(2~6): 하위 브랜드들
                For j = 2 To 6
                    Dim subBrandName As String
                    subBrandName = Trim(CStr(ws.Cells(i, j).value))
                    
                    If subBrandName <> "" Then
                        subBrands.Add subBrandName
                        If debugMode Then Debug.Print "    하위브랜드 추가: " & subBrandName
                    End If
                Next j
                
                If subBrands.Count > 0 Then
                    ' ?? 핵심 수정: Dictionary에 값 설정
                    brandItem.Add "Name", brandName
                    brandItem.Add "Brands", subBrands
                    
                    ' ?? 핵심 수정: Collection에 추가
                    result.Add brandItem
                    
                    If debugMode Then Debug.Print "  → ? 통합브랜드 추가 완료: " & brandName & " (" & subBrands.Count & "개)"
                Else
                    If debugMode Then Debug.Print "  → ? 하위브랜드 없음 - 제외"
                End If
            Else
                If debugMode Then Debug.Print "  → ? 브랜드명 비어있음"
            End If
        Else
            If debugMode Then Debug.Print "  → ? 사용안함 (G열: " & useFlag & ")"
        End If
    Next i
    
    If debugMode Then
        Debug.Print ""
        Debug.Print "=== LoadIntegratedBrands 완료 ==="
        Debug.Print "총 통합브랜드: " & result.Count & "개"
        Debug.Print "================================" & vbCrLf
    End If
    
    Set LoadIntegratedBrands = result
    Exit Function
    
ErrorHandler:
    If debugMode Then
        Debug.Print "? 오류 발생!"
        Debug.Print "Error " & Err.Number & ": " & Err.Description
        Debug.Print "오류 발생 위치: " & Erl
    End If
    Set LoadIntegratedBrands = New Collection
End Function

' 브랜드 → 통합브랜드 매핑
Public Function GetIntegratedBrandName(brandName As String) As String
    On Error GoTo ErrorHandler
    
    Dim ws As Worksheet
    Dim lastRow As Long
    Dim i As Long, j As Long
    Dim sheetFound As Boolean
    
    ' 시트 찾기
    sheetFound = False
    
    On Error Resume Next
    Set ws = ThisWorkbook.Sheets("브랜드설정")
    If Not ws Is Nothing Then sheetFound = True
    On Error GoTo ErrorHandler
    
    If Not sheetFound Then
        On Error Resume Next
        Set ws = ThisWorkbook.Sheets("브랜드 설정")
        If Not ws Is Nothing Then sheetFound = True
        On Error GoTo ErrorHandler
    End If
    
    If Not sheetFound Then
        On Error Resume Next
        Set ws = ThisWorkbook.Sheets(SHEET_브랜드설정)
        If Not ws Is Nothing Then sheetFound = True
        On Error GoTo ErrorHandler
    End If
    
    If Not sheetFound Or ws Is Nothing Then
        GetIntegratedBrandName = brandName
        Exit Function
    End If
    
    lastRow = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row
    
    For i = 2 To lastRow
        If ws.Cells(i, 7).value = "O" Then
            For j = 2 To 6
                If ws.Cells(i, j).value = brandName Then
                    GetIntegratedBrandName = ws.Cells(i, 1).value
                    Exit Function
                End If
            Next j
        End If
    Next i
    
    ' 통합 브랜드에 속하지 않으면 원래 이름
    GetIntegratedBrandName = brandName
    Exit Function
    
ErrorHandler:
    GetIntegratedBrandName = brandName
End Function
